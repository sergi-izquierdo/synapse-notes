\chapter*{Glossari de sigles}
\addcontentsline{toc}{chapter}{Glossari de sigles}

\begin{description}[leftmargin=4.5cm,style=nextline]
    \item[ADA] Anàlisi i Disseny d'Aplicacions (assignatura URV).
    \item[ANN] Approximate Nearest Neighbor (cerca vectorial aproximada).
    \item[API] Application Programming Interface.
    \item[BFS] Breadth-First Search.
    \item[BD] Base de Dades.
    \item[CRUD] Create, Read, Update, Delete.
    \item[D1-D5] Identificadors de les decisions arquitectòniques majors del projecte (registre a \texttt{docs/tfg/00-decision-log.md}).
    \item[DCR] Dynamic Client Registration (RFC~7591).
    \item[E2EE] End-to-End Encryption.
    \item[F1] Mètrica F1, mitjana harmònica de precision i recall.
    \item[FK] Foreign Key.
    \item[GIN] Generalized Inverted Index (índex Postgres per a cerca textual).
    \item[HNSW] Hierarchical Navigable Small World (índex vectorial gràfic).
    \item[HTTPS] HyperText Transfer Protocol Secure.
    \item[IA] Intel·ligència Artificial.
    \item[IdP] Identity Provider.
    \item[JSON-RPC] JSON Remote Procedure Call (protocol RPC sobre JSON).
    \item[JWT] JSON Web Token.
    \item[LLM] Large Language Model.
    \item[LOPDGDD] Llei Orgànica 3/2018 de Protecció de Dades i garantia dels Drets Digitals.
    \item[LSSI-CE] Llei de Serveis de la Societat de la Informació i Comerç Electrònic (Llei 34/2002).
    \item[MCP] Model Context Protocol (estàndard d'Anthropic per a integració d'eines amb LLMs).
    \item[OAuth] Open Authorization (estàndard d'autorització delegada).
    \item[PKCE] Proof Key for Code Exchange (RFC~7636).
    \item[PoC] Proof of Concept.
    \item[RAG] Retrieval-Augmented Generation.
    \item[RF] Requisit Funcional.
    \item[RFC] Request for Comments (estàndard IETF).
    \item[RGPD] Reglament General de Protecció de Dades (Reglament UE 2016/679).
    \item[RLS] Row-Level Security (mecanisme d'autorització de fila a PostgreSQL).
    \item[RLHF] Reinforcement Learning from Human Feedback.
    \item[RNF] Requisit No Funcional.
    \item[RPC] Remote Procedure Call.
    \item[SaaS] Software as a Service.
    \item[SCC] Standard Contractual Clauses (clàusules tipus UE per a transferències internacionals).
    \item[SDK] Software Development Kit.
    \item[SSO] Single Sign-On.
    \item[TFG] Treball de Fi de Grau.
    \item[TLS] Transport Layer Security.
    \item[TTL] Time To Live.
    \item[UI] User Interface.
    \item[URV] Universitat Rovira i Virgili.
    \item[UUID] Universally Unique Identifier.
    \item[WCAG] Web Content Accessibility Guidelines.
\end{description}
